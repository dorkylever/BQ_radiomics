# Global pip constraints for the container.
# The scientific stack is provided by conda and is ABI-locked. Capping numpy < 2
# stops any `pip install` — including the editable project install and its
# unpinned deps like `shap` — from pulling numpy 2.x, which is incompatible with
# numba and the pyradiomics 3.1.0 wheel.
numpy>=1.23,<2
